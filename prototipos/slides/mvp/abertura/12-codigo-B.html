def novo_preprocessador():
    """Codificação das categóricas e padronização
    das numéricas."""
    return ColumnTransformer([
        ('cat', Pipeline([
            ('imp', SimpleImputer(strategy='constant',
                                 fill_value='(vazio)')),
            ('ohe', OneHotEncoder(handle_unknown='ignore',
                                 min_frequency=100)),
        ]), CATEGORICAS),
        ('num', Pipeline([
            ('imp', SimpleImputer(strategy='median')),
            ('sc', StandardScaler()),
        ]), NUMERICAS),
    ])

modelo = Pipeline([
    ('pre', novo_preprocessador()),
    ('clf', LogisticRegression(max_iter=5000,
                             random_state=SEMENTE))])
modelo.fit(X_treino, y_treino)
risco_teste = modelo.predict_proba(X_teste)[:, 1]

# metas anuais de violação de OLA, do dicionário
# de dados oficial: (teto de quebras, atingimento)
ESCADA = {
    'P2': [(30, 150), (35, 125), (39, 100),
           (45, 75), (53, 50), (float('inf'), 0)],
    'P3': [(200, 150), (230, 125), (263, 100),
           (290, 75), (320, 50), (float('inf'), 0)],
}

def atingimento(pri, quebras):
    """Em que degrau da escada do KPI aquele
    número de quebras cai."""
    degraus = ESCADA[pri]
    for i, (teto, pc) in enumerate(degraus):
        if quebras <= teto:
            return {
                'pc': pc,
                'margem': round(teto - quebras, 1),
                'proximo_pc': degraus[i + 1][1],
            }
